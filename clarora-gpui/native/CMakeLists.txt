cmake_minimum_required(VERSION 3.20)
project(clarora_gpui_asr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
# Reuse the project's public native ASR adapter, without any React Native bridge.
set(ASR_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../clarora-app/windows/clarora-asr")
add_executable(clarora-asr asr_main.cpp "${ASR_SOURCE}/clarora_asr.cpp")
target_compile_definitions(clarora-asr PRIVATE MINIAUDIO_IMPLEMENTATION MA_NO_DEVICE_IO)
find_path(WHISPER_INCLUDE whisper.h REQUIRED)
find_library(WHISPER_LIBRARY whisper REQUIRED)
find_path(GGML_INCLUDE ggml-backend.h REQUIRED)
find_library(GGML_LIBRARY ggml REQUIRED)
get_filename_component(GGML_REAL_LIBRARY "${GGML_LIBRARY}" REALPATH)
get_filename_component(GGML_LIB_DIR "${GGML_REAL_LIBRARY}" DIRECTORY)
find_file(GGML_CPU_BACKEND NAMES libggml-cpu.so libggml-cpu.dylib ggml-cpu.dll
    PATHS "${GGML_LIB_DIR}/../libexec" "${GGML_LIB_DIR}" "${GGML_LIB_DIR}/../bin")
if(GGML_CPU_BACKEND)
  target_compile_definitions(clarora-asr PRIVATE CLARORA_GGML_CPU_BACKEND="${GGML_CPU_BACKEND}")
endif()
find_path(SHERPA_INCLUDE sherpa-onnx/c-api/c-api.h REQUIRED)
find_library(SHERPA_LIBRARY sherpa-onnx-c-api REQUIRED)
target_include_directories(clarora-asr PRIVATE "${ASR_SOURCE}" "${WHISPER_INCLUDE}" "${GGML_INCLUDE}" "${SHERPA_INCLUDE}")
target_link_libraries(clarora-asr PRIVATE "${WHISPER_LIBRARY}" "${GGML_LIBRARY}" "${SHERPA_LIBRARY}")
if(APPLE)
  target_link_libraries(clarora-asr PRIVATE "-framework CoreFoundation" "-framework AudioToolbox")
endif()
install(TARGETS clarora-asr RUNTIME DESTINATION tools)
