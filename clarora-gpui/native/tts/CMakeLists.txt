cmake_minimum_required(VERSION 3.20)
project(clarora_gpui_tts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_path(SHERPA_INCLUDE sherpa-onnx/c-api/c-api.h REQUIRED)
find_library(SHERPA_LIBRARY sherpa-onnx-c-api REQUIRED)
add_executable(clarora-tts ../tts_main.cpp)
target_include_directories(clarora-tts PRIVATE "${SHERPA_INCLUDE}")
target_link_libraries(clarora-tts PRIVATE "${SHERPA_LIBRARY}")
install(TARGETS clarora-tts RUNTIME DESTINATION tools)
if(APPLE)
  # Keep TTS libraries beside its helper, separate from an ASR-only installation.
  set_target_properties(clarora-tts PROPERTIES BUILD_RPATH "@loader_path" INSTALL_RPATH "@loader_path")
  get_filename_component(TTS_LIBRARY_DIR "${SHERPA_LIBRARY}" DIRECTORY)
  file(GLOB TTS_RUNTIME_LIBS "${TTS_LIBRARY_DIR}/libsherpa-onnx-c-api*.dylib" "${TTS_LIBRARY_DIR}/libonnxruntime*.dylib")
  foreach(runtime IN LISTS TTS_RUNTIME_LIBS)
    add_custom_command(TARGET clarora-tts POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${runtime}" "$<TARGET_FILE_DIR:clarora-tts>/")
  endforeach()
endif()
